Importação dos dados de saída e entrada

load("DataIdentboost.mat")
Nd = 300;
t = DadosBoost(1,1:end-Nd);

v = DadosBoost(3,Nd:end-1);

mu_u = mean(u);
mu_v = mean(v);

u = u - mu_u;
v = v - mu_v;
y = transpose(v);

Construção matricial - regressores

Ns = numel(t);

M = zeros(Ns-3,4);

for n = 3:Ns
    phi = [y(n-1), y(n-2), u(n-1), u(n-2)];
    M(n, :) = phi;
end

theta = M\y;
a1 = theta(1);
a2 = theta(2);
b1 = theta(3);
b2 = theta(4);
ys = zeros(size(y));
ys(1:2) = y(1:2);

yp = M * theta;
