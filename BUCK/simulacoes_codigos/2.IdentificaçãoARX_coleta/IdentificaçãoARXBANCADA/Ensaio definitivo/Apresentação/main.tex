\section{Fundamentação Teórica}
%-------------------------------------------------------
\begin{frame}{Fundamentação Teórica}{Coleta de Dados e Modelo ARX}
\begin{block}{Coleta de Variáveis}
    \begin{itemize}
         \item \textbf{Entrada ($u(k)$):} Sinal PRBS (Pseudo-Random Binary Sequence) aplicado no gate da chave MOSFET.
         \item \textbf{Saída ($y(k)$):} Tensão dissipada na carga resistiva.
    \end{itemize} 
\end{block}

\begin{block}{Estrutura do Modelo ARX}
    \begin{itemize}
         \item O modelo ARX (AutoRegressive with eXogenous input) descreve a relação linear do sistema através de uma equação de diferenças.
         \item Equação geral: $y(k) = \sum_{i=1}^{n} a_i y(k-i) + \sum_{j=1}^{n} b_j u(k-j)$.
         \item O vetor de parâmetros $X$ é estimado pelo método dos Mínimos Quadrados: $X = (A^T \cdot A)^{-1} \cdot A^T \cdot Y$, onde $A$ é a matriz de regressores.
    \end{itemize}
\end{block}
\end{frame}

\begin{frame}{Análise dos Resultados}{Funções de Transferência e Desempenho}
\textbf{Funções de Transferência Discretas ($50\,\mu\text{s}$)}\\
\vspace{5pt}
\begin{itemize}
    \item \textbf{Modelo Analítico Ideal ($G_z$):}
    $$ G_z = \frac{1.785z + 1.785}{z^2 - 1.768z + 0.8869} $$
    \item \textbf{Modelo Estimado ($G_{ze}$):}
    $$ G_{ze} = \frac{0.01594z - 0.03983}{z^2 - 1.729z + 0.7917} $$
\end{itemize}

\vspace{10pt}
\textbf{Validação do Ajuste}\\
\begin{itemize}
    \item O Coeficiente de Determinação foi de $R^2 = 0.9741$.
    \item Indica que o modelo estimado representa $97.41\%$ de todas as variações contidas na saída real da planta física.
\end{itemize}
\end{frame}

\section{Conclusão}
%-------------------------------------------------------
\begin{frame}{Conclusão}{Considerações Finais}
  \begin{block}{Conclusão}
\begin{itemize}
    \item A estrutura de modelo ARX demonstrou excelente aderência para a identificação linear do conversor Buck operando sob excitação de sinal PRBS.
    \vspace{5pt}
    \item O modelo extraído validou a eficácia do método dos mínimos quadrados com uma representatividade de $97.41\%$ (Coeficiente $R^2$).

\end{itemize}
  \end{block}
